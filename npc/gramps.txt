//===== How to customize: ====================================
//= - Per bracket (7 of them): the callsub OnHandleQuest line sets
//=   min/max level range, monster IDs, kill counts needed, and EXP reward.
//=   -- but the kill count (400) and monster IDs are ALSO checked
//=   separately in OnTIKill below, so change both places together.
//= - Level requirements: minimum levels appear twice per bracket -- 
//=   once in the warp menu, once in OnHandleQuest -- keep both in sync.
//= - Monster counts on the maps themselves: gramps_mobs.txt, not
//=   here.
//= - Warp destinations: the warp lines under "MAP WARP LOGIC".
//= - Progress is tracked per CHARACTER (Gramps_Kill1/2,
//=   Gramps_ActiveBracket, Gramps_PenaltyFlag -- no # prefix).
//= - Players can take lower-tier bounties if above max level, but
//=   accepting an over-level bounty flags Gramps_PenaltyFlag = 1,
//=   reducing EXP payout to 1/4th. Taking a bounty ON-LEVEL and
//=   leveling past max during the hunt keeps Gramps_PenaltyFlag = 0
//=   and yields 100% full EXP reward.
//= - Ongoing bounties can be abandoned via the dynamic main menu option
//=   with target details explicitly highlighted in the confirmation menu.
//============================================================

prontera,147,172,5	script	Gramps	843,{
    .@base_lv = BaseLevel;

    mes "[Gramps]";
    mes "Hello there, adventurer! The Eden Group keeps me busy coordinating hunting bounties for mercenaries.";
    mes "I have level-bracketed Turn-In bounties available. You can even take lower-tier bounties if you prefer, though over-level bounties grant reduced EXP!";
    next;

    // Build dynamic menu based on active bounty state
    if (Gramps_ActiveBracket != 0) {
        .@quest_option$ = "Check / Turn-In Bounty:Abandon Ongoing Bounty";
    } else {
        .@quest_option$ = "Check / Take Bounties";
    }

    .@menu = select(.@quest_option$ + ":Warp to Hunting Map:Cancel");

    // ABANDON BOUNTY LOGIC
    if (Gramps_ActiveBracket != 0 && .@menu == 2) {
        // Resolve active bounty details for confirmation
        switch(Gramps_ActiveBracket) {
            case 1: .@b_name$ = "Lv 70-100 (Clock & Anolian)"; .@m1$ = "Clock"; .@m2$ = "Anolian"; break;
            case 2: .@b_name$ = "Lv 101-125 (Kaho & Flame Shooter)"; .@m1$ = "Kaho"; .@m2$ = "Flame Shooter"; break;
            case 3: .@b_name$ = "Lv 126-150 (Horn & Antler Scaraba)"; .@m1$ = "Horn Scaraba"; .@m2$ = "Antler Scaraba"; break;
            case 4: .@b_name$ = "Lv 151-175 (Corrupted Robots)"; .@m1$ = "Corrupted Repair Robot"; .@m2$ = "Corrupted Exploration Rover"; break;
            case 5: .@b_name$ = "Lv 176-200 (Corrupted Mine Mobs)"; .@m1$ = "Corrupted Jewelry Ant"; .@m2$ = "Corrupted Abyssman"; break;
            case 6: .@b_name$ = "Lv 201-230 (Corrupted Niflheim Mobs)"; .@m1$ = "Corrupted Murderer"; .@m2$ = "Corrupted Ghost Cube"; break;
            case 7: .@b_name$ = "Lv 231+ (Corrupted Niflheim Mobs, Deep)"; .@m1$ = "Corrupted Murderer"; .@m2$ = "Corrupted Ghost Cube"; break;
        }

        mes "[Gramps]";
        mes "Are you sure you want to give up on your current active bounty?";
        mes "Bounty: ^0055FF" + .@b_name$ + "^000000";
        mes "Progress: " + .@m1$ + " (" + Gramps_Kill1 + "/400), " + .@m2$ + " (" + Gramps_Kill2 + "/400)";
        mes "\n^FF0000All progress on this bounty will be permanently lost!^000000";
        next;
        if (select("No, keep bounty:Yes, abandon bounty") == 2) {
            Gramps_ActiveBracket = 0;
            Gramps_Kill1 = 0;
            Gramps_Kill2 = 0;
            Gramps_PenaltyFlag = 0;
            mes "[Gramps]";
            mes "Your active bounty for ^0055FF" + .@b_name$ + "^000000 has been abandoned. You are now free to take on a new contract whenever you're ready.";
            close;
        }
        mes "[Gramps]";
        mes "Good focus! Get back out there and finish the job.";
        close;
    }

    // Shift menu indexing if "Abandon Bounty" option was rendered
    if (Gramps_ActiveBracket != 0 && .@menu > 2) {
        .@menu -= 1;
    }

    // CANCEL OPTION
    if (.@menu == 3) {
        mes "[Gramps]";
        mes "Stay safe out there! Come back when you're ready to hunt.";
        close;
    }

    // MAP WARP LOGIC
    if (.@menu == 2) {
        mes "[Gramps]";
        mes "Where would you like to be deployed?";
        next;
        .@warp_choice = select("Level 70+ Ground (alde_dun04):Level 101+ Ground (mag_dun02):Level 126+ Ground (dic_dun02):Level 151+ Ground (verus03):Level 176+ Ground (ein_dun03):Level 201+ Ground (nif_dun01):Level 231+ Ground (nif_dun02):Cancel");

        switch(.@warp_choice) {
            case 1:
                if (.@base_lv < 70) {
                    mes "[Gramps]";
                    mes "You're too weak for this ground! Minimum required level is 70.";
                    close;
                }
                warp "alde_dun04", 103, 162;
                end;
            case 2:
                if (.@base_lv < 101) {
                    mes "[Gramps]";
                    mes "You need to be at least Level 101 for this ground!";
                    close;
                }
                warp "mag_dun02", 46, 49;
                end;
            case 3:
                if (.@base_lv < 126) {
                    mes "[Gramps]";
                    mes "You need to be at least Level 126 for this ground!";
                    close;
                }
                warp "dic_dun02", 100, 100;
                end;
            case 4:
                if (.@base_lv < 151) {
                    mes "[Gramps]";
                    mes "You need to be at least Level 151 for this ground!";
                    close;
                }
                warp "verus03", 100, 100;
                end;
            case 5:
                if (.@base_lv < 176) {
                    mes "[Gramps]";
                    mes "You need to be at least Level 176 for this ground!";
                    close;
                }
                warp "ein_dun03", 100, 100;
                end;
            case 6:
                if (.@base_lv < 201) {
                    mes "[Gramps]";
                    mes "You need to be at least Level 201 for this ground!";
                    close;
                }
                warp "nif_dun01", 100, 100;
                end;
            case 7:
                if (.@base_lv < 231) {
                    mes "[Gramps]";
                    mes "You need to be at least Level 231 for this ground!";
                    close;
                }
                warp "nif_dun02", 100, 100;
                end;
            case 8:
                close;
        }
    }

    // TURN-IN BOUNTY LOGIC
    mes "[Gramps]";
    mes "Select a level bracket suitable for your current Base Level:";
    next;

    .@bracket = select("Level 70 - 100:Level 101 - 125:Level 126 - 150:Level 151 - 175:Level 176 - 200:Level 201 - 230:Level 231+:Cancel");

    switch(.@bracket) {
        case 1: // Lv 70 - 100 Bracket (Clock & Anolian)
            callsub OnHandleQuest, 1, 70, 100, 1269, 1206, 400, 400, 1500000, 1000000, "Clock", "Anolian";
            break;

        case 2: // Lv 101 - 125 Bracket (Kaho & Flame Shooter)
            callsub OnHandleQuest, 2, 101, 125, 1131, 1368, 400, 400, 4500000, 3000000, "Kaho", "Flame Shooter";
            break;

        case 3: // Lv 126 - 150 Bracket (Horn Scaraba & Antler Scaraba)
            callsub OnHandleQuest, 3, 126, 150, 2082, 2083, 400, 400, 12000000, 8000000, "Horn Scaraba", "Antler Scaraba";
            break;

        case 4: // Lv 151 - 175 Bracket (Corrupted Repair Robot & Corrupted Exploration Rover)
            callsub OnHandleQuest, 4, 151, 175, 3248, 3249, 400, 400, 25000000, 18000000, "Corrupted Repair Robot", "Corrupted Exploration Rover";
            break;

        case 5: // Lv 176 - 200 Bracket (Corrupted Jewelry Ant & Corrupted Abyssman)
            callsub OnHandleQuest, 5, 176, 200, 20598, 20603, 400, 400, 50000000, 35000000, "Corrupted Jewelry Ant", "Corrupted Abyssman";
            break;

        case 6: // Lv 201 - 230 Bracket (Corrupted Murderer & Corrupted Ghost Cube)
            callsub OnHandleQuest, 6, 201, 230, 20937, 20938, 400, 400, 100000000, 75000000, "Corrupted Murderer", "Corrupted Ghost Cube";
            break;

        case 7: // Lv 231+ Bracket (Corrupted Murderer & Corrupted Ghost Cube, deep Niflheim)
            callsub OnHandleQuest, 7, 231, 999, 20937, 20938, 400, 400, 150000000, 100000000, "Corrupted Murderer", "Corrupted Ghost Cube";
            break;

        case 8:
            close;
    }
    end;

// Subroutine: Bounty Handling Framework
OnHandleQuest:
    .@bracket_id = getarg(0);
    .@min_lv     = getarg(1);
    .@max_lv     = getarg(2);
    .@m1         = getarg(3); .@m2 = getarg(4);
    .@req1       = getarg(5); .@req2 = getarg(6);
    .@base_exp   = getarg(7); .@job_exp = getarg(8);
    .@mob1_name$ = getarg(9); .@mob2_name$ = getarg(10);

    // Minimum Level Check
    if (BaseLevel < .@min_lv) {
        mes "[Gramps]";
        mes "Your level (" + BaseLevel + ") is below the minimum Level " + .@min_lv + " required for this bounty!";
        close;
    }

    if (Gramps_ActiveBracket != 0 && Gramps_ActiveBracket != .@bracket_id) {
        mes "[Gramps]";
        mes "You already have an active bounty in another level bracket! Finish or abandon it first.";
        close;
    }

    // REWARD TURN-IN CHECK
    if (Gramps_Kill1 >= .@req1 && Gramps_Kill2 >= .@req2) {
        
        // Calculate penalty if bounty was TAKEN when player was higher than max_lv
        if (Gramps_PenaltyFlag == 1) {
            .@base_exp = .@base_exp / 4;
            .@job_exp  = .@job_exp / 4;
            .@penalty_note$ = " ^FF0000(Reduced EXP: Over-level Bounty)^000000";
        } else {
            .@penalty_note$ = "";
        }

        Gramps_Kill1 = 0;
        Gramps_Kill2 = 0;
        Gramps_ActiveBracket = 0;
        Gramps_PenaltyFlag = 0;

        getexp .@base_exp, .@job_exp;
        specialeffect2 154;
        mes "[Gramps]";
        mes "Outstanding job! Here is your reward" + .@penalty_note$ + ":";
        mes "^0055FF+" + .@base_exp + " Base EXP^000000";
        mes "^0055FF+" + .@job_exp + " Job EXP^000000";
        close;
    }

    // ALREADY ACCEPTED CHECK
    if (Gramps_ActiveBracket == .@bracket_id) {
        mes "[Gramps]";
        mes "You still have active bounties for this bracket! Current progress:";
        mes "- " + .@mob1_name$ + ": ^0055FF" + Gramps_Kill1 + "^000000 / " + .@req1;
        mes "- " + .@mob2_name$ + ": ^0055FF" + Gramps_Kill2 + "^000000 / " + .@req2;
        if (Gramps_PenaltyFlag == 1) {
            mes "^FF0000* Note: This bounty was accepted above the recommended level range and will reward 1/4th EXP.^000000";
        }
        close;
    }

    // ACCEPTING NEW BOUNTY
    Gramps_ActiveBracket = .@bracket_id;
    Gramps_Kill1 = 0;
    Gramps_Kill2 = 0;

    // Set penalty flag ONLY if accepted while ALREADY higher than the upper bracket limit
    if (BaseLevel > .@max_lv) {
        Gramps_PenaltyFlag = 1;
        .@warning$ = "\n\n^FF0000Warning: Because your level is above Level " + .@max_lv + ", this bounty will reward 25% EXP upon turn-in.^000000";
    } else {
        Gramps_PenaltyFlag = 0;
        .@warning$ = "";
    }

    mes "[Gramps]";
    mes "Bounty accepted! Defeat " + .@req1 + "x ^0055FF" + .@mob1_name$ + "^000000 and " + .@req2 + "x ^0055FF" + .@mob2_name$ + "^000000, then return to me for your EXP reward." + .@warning$;
    close;
}
