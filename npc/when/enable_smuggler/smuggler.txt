//===== rAthena Script =======================================
//= Smuggler
//===== Description: =========================================
//= Free-form buy: type any item ID, get it at a markup. No
//= restrictions (private server) -- means ANY item in item_db
//= can be bought this way, event/GM/quest items included.
//===== How to customize: ====================================
//= - Which item types get the flat high price (no markup):
//=   .@fixed_types below, values from getiteminfo(id,
//=   ITEMINFO_TYPE) -- currently 0 Healing, 2 Usable, 4 Armor,
//=   5 Weapon, 6 Card, 7 Pet Egg, 8 Pet Armor, 10 Ammo,
//=   11 Delay-consume, 12 Shadow Gear, 18 Cash.
//= - Flat price for those types: .@fixed_price below (10,000,000).
//= - Everything else (currently types 1, 3, 9 -- Unknown, Etc/
//=   quest items, Unknown2 -- plus anything not in the list
//=   above): .@fallback (1000z) when item_db's own Buy price is
//=   0, then .@markup (x5) on top.
//= - Location/sprite: the first line below.
//============================================================

prontera,144,174,4	script	Smuggler	494,{
	mes "[Smuggler]";
	mes "For the right price, I can get you anything... if I have a supplier for it.";
	mes "Item ID?";
	next;
	input .@id;
	.@price = getiteminfo(.@id,ITEMINFO_BUY);
	if (.@price < 0) {
		mes "[Smuggler]";
		mes "Never heard of it. No deal.";
		close;
	}
	.@type = getiteminfo(.@id,ITEMINFO_TYPE);
	setarray .@fixed_types[0], 0,2,4,5,6,7,8,10,11,12,18;
	if (inarray(.@fixed_types, .@type) != -1) {
		.@fixed_price = 10000000;
		.@price = .@fixed_price;
	} else {
		if (.@price == 0) {
			.@fallback = 1000;
			.@price = .@fallback;
		}
		.@markup = 5;
		.@price = .@price * .@markup;
	}
	mes "[Smuggler]";
	mes "^0000FF"+getitemname(.@id)+"^000000 -- "+.@price+"z. Deal?";
	next;
	if (select("Deal:Never mind") == 2) close;
	if (Zeny < .@price) {
		mes "[Smuggler]";
		mes "Come back with more money.";
		close;
	}
	if (!checkweight(.@id,1)) {
		mes "[Smuggler]";
		mes "You can't carry that right now.";
		close;
	}
	Zeny -= .@price;
	getitem .@id,1;
	mes "[Smuggler]";
	mes "Pleasure doing business.";
	close;
}
