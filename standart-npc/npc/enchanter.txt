//===== rAthena Script =======================================
//= Costume Stone Enchanter (Class Stones Enabled)
//===== By: Lil Art ===========================================
//===== How to customize: ====================================
//= - Free -- no Zeny cost, just consumes the card/stone used.
//= - Which items count as a valid stone: the ID range check
//=   (25000-32000, and 100000+) plus "is a card" -- widen/narrow
//=   that range to change what's accepted.
//= - Location/sprite: the first line (map,x,y,dir / sprite).
//============================================================

prontera,164,169,4	script	Costume Enchanter	55,{
    mes "[Costume Enchanter]";
    mes "Greetings! I can attach any ^0055FFEnchant Stone, Class Stone, or Card^000000 directly to your equipped costume headgears!";
    next;

    if (select("Enchant Costume Headgear:Cancel") == 2) close;

    // Step 1: Select Costume Slot
    mes "[Costume Enchanter]";
    mes "Which equipped costume piece do you want to enchant?";
    next;

    setarray .@EQ_SLOTS[1], EQI_COSTUME_HEAD_TOP, EQI_COSTUME_HEAD_MID, EQI_COSTUME_HEAD_LOW;
    setarray .@EQ_NAMES$[1], "Costume Top Headgear", "Costume Mid Headgear", "Costume Low Headgear";

    .@menu_str$ = "";
    for (.@i = 1; .@i <= getarraysize(.@EQ_SLOTS); .@i++) {
        .@eq_id = getequipid(.@EQ_SLOTS[.@i]);
        if (.@eq_id > 0) {
            .@menu_str$ += .@EQ_NAMES$[.@i] + " [" + getitemname(.@eq_id) + "]";
        } else {
            .@menu_str$ += .@EQ_NAMES$[.@i] + " [Empty]";
        }
        .@menu_str$ += ":";
    }
    .@menu_str$ += "Cancel";

    .@sel_slot = select(.@menu_str$);
    if (.@sel_slot > getarraysize(.@EQ_SLOTS)) close;

    .@target_eq = .@EQ_SLOTS[.@sel_slot];
    .@target_id = getequipid(.@target_eq);

    if (.@target_id <= 0) {
        mes "[Costume Enchanter]";
        mes "You do not have a costume equipped in that slot!";
        close;
    }

    // Step 2: Select Target Socket Slot (0 to 3)
    mes "[Costume Enchanter]";
    mes "Selected: ^0055FF" + getitemname(.@target_id) + "^000000";
    mes "Which socket do you want to enchant?";
    next;

    .@c0 = getequipcardid(.@target_eq, 0);
    .@c1 = getequipcardid(.@target_eq, 1);
    .@c2 = getequipcardid(.@target_eq, 2);
    .@c3 = getequipcardid(.@target_eq, 3);

    .@sock_str$ = "Socket 1 [" + (.@c0 ? getitemname(.@c0) : "^00FF00Empty^000000") + "]:"
                + "Socket 2 [" + (.@c1 ? getitemname(.@c1) : "^00FF00Empty^000000") + "]:"
                + "Socket 3 [" + (.@c2 ? getitemname(.@c2) : "^00FF00Empty^000000") + "]:"
                + "Socket 4 [" + (.@c3 ? getitemname(.@c3) : "^00FF00Empty^000000") + "]:Cancel";

    .@sock_choice = select(.@sock_str$) - 1;
    if (.@sock_choice >= 4) close;

    if (getequipcardid(.@target_eq, .@sock_choice) > 0) {
        mes "[Costume Enchanter]";
        mes "That socket is already occupied!";
        close;
    }

    // Step 3: Expanded Inventory Scanner (Cards + Class/Enchant Stones)
    getinventorylist;
    .@stone_count = 0;
    deletearray .@valid_stone_id[0];

    for (.@i = 0; .@i < @inventorylist_count; .@i++) {
        .@id = @inventorylist_id[.@i];
        .@type = @inventorylist_type[.@i];
        
        // Check if item is a Card OR falls in the official Enchant/Class Stone ID ranges (25000-32000 & 100000+)
        if (.@type == IT_CARD || ( .@id >= 25000 && .@id <= 32000 ) || .@id >= 100000) {
            .@valid_stone_id[.@stone_count] = .@id;
            .@stone_count++;
        }
    }

    if (.@stone_count == 0) {
        mes "[Costume Enchanter]";
        mes "You do not have any Cards, Class Stones, or Enchant Stones in your inventory!";
        close;
    }

    // Step 4: Choose Stone
    mes "[Costume Enchanter]";
    mes "Select the Class Stone or Enchantment to attach:";
    next;

    .@stone_menu$ = "";
    for (.@i = 0; .@i < .@stone_count; .@i++) {
        .@stone_menu$ += getitemname(.@valid_stone_id[.@i]) + ":";
    }
    .@stone_menu$ += "Cancel";

    .@stone_sel = select(.@stone_menu$) - 1;
    if (.@stone_sel >= .@stone_count) close;

    .@chosen_stone_id = .@valid_stone_id[.@stone_sel];

    mes "[Costume Enchanter]";
    mes "Attach ^0055FF" + getitemname(.@chosen_stone_id) + "^000000 to ^0055FF" + getitemname(.@target_id) + "^000000?";
    next;

    if (select("Yes, proceed:Cancel") == 2) close;

    if (countitem(.@chosen_stone_id) < 1) {
        mes "[Costume Enchanter]";
        mes "You are missing the required stone!";
        close;
    }

    // Process Enchanting
    .@cards[0] = .@c0;
    .@cards[1] = .@c1;
    .@cards[2] = .@c2;
    .@cards[3] = .@c3;
    .@cards[.@sock_choice] = .@chosen_stone_id;

    delitem .@chosen_stone_id, 1;
    delequip .@target_eq;
    getitem2 .@target_id, 1, 1, 0, 0, .@cards[0], .@cards[1], .@cards[2], .@cards[3];
    equip .@target_id;

    specialeffect2 154;
    mes "[Costume Enchanter]";
    mes "Enchantment successful!";
    close;
}
